```rust
use anyhow::anyhow;
use common::models::session_entity::user_session::UserSession;
use rbatis::rbdc::Uuid;
use rbatis::RBatis;
use tracing::{info, warn};

use crate::http_service::session_service::dto::{SessionReadDTO, SessionSyncedDTO};

/// 已读上报(§7.1): 逐条推进, 钳制与只前进在数据层保证(任务03)。
/// 非法 session_uuid 记 warn 跳过, 不整体失败(部分成功优于全部拒绝)。
pub async fn report_session_read(
    rb: &RBatis,
    me: &str,
    dto: SessionReadDTO,
) -> Result<(), anyhow::Error> {
    let me: Uuid = me.parse()?;
    for item in dto.reads {
        let Ok(session_uuid) = item.session_uuid.parse::<Uuid>() else {
            warn!("[session/read] 非法 session_uuid: {}", item.session_uuid);
            continue;
        };
        let affected = UserSession::update_last_read_id(
            rb, &me, &session_uuid, item.last_read_id,
        ).await?;
        if affected == 0 {
            // 行不存在(未聚合)或无需推进 —— 前者记日志便于排查
            warn!("[session/read] 无效上报: user={}, session={}", me, session_uuid);
        }
    }
    info!("[session/read] 上报完成: user={}, {} 条", me, dto.reads.len());
    Ok(())
}

/// 同步游标回报(§7.3): 逐条推进, 只前进。
pub async fn report_session_synced(
    rb: &RBatis,
    me: &str,
    dto: SessionSyncedDTO,
) -> Result<(), anyhow::Error> {
    let me: Uuid = me.parse()?;
    for item in dto.sessions {
        let Ok(session_uuid) = item.session_uuid.parse::<Uuid>() else {
            warn!("[session/synced] 非法 session_uuid: {}", item.session_uuid);
            continue;
        };
        let _ = UserSession::update_synced_id(rb, &me, &session_uuid, item.synced_id).await?;
    }
    info!("[session/synced] 回报完成: user={}, {} 条", me, dto.sessions.len());
    Ok(())
}
```

## 4. 桥接：`/msg/add_read_chat_record` 内部重写

**契约不变**（`Vec<AddReadChatRecordDTO>`），内部从「写 Redis 等 QUIC 上线同步」改为「直写 `user_session` 游标」。客户端零改动（缺陷 J 消除：不再有 24h TTL 丢失窗口）。

### 4.1 语义转换

旧 DTO 携带的是「读到的消息」（nano_id 级）；新游标是会话级的。转换规则 = **读到底**（该会话当前最大消息 id）：

> 用户停留在聊天界面看到某条消息时，语义上就是"看到了该会话此刻的最新"—— 推到 max(id) 是正确近似，且免去按 nano_id 反查消息（分区表跨分区扫，单聊侧连索引都没有）。

### 4.2 `text_msg_service.rs` 的 `add_user_chat_read` 重写

```rust
/// 用户已读上报(桥接): 旧 DTO 契约保留, 内部直写 user_session 游标。
/// Redis 缓冲路径删除(缺陷J: 24h TTL 丢失窗口)。
pub async fn add_user_chat_read(
    rb: &RBatis,
    me: &str,
    chat_message_read: Vec<AddReadChatRecordDTO>,
) -> Result<String, anyhow::Error> {
    let me_uuid: Uuid = me.parse()?;
    for item in chat_message_read {
        if item.chat_type == Some(CHAT_TYPE_GROUP) {
            // 群聊: nano_id 反查群消息拿 group_uuid(旧路径同款, 频率低可接受)
            let group_uuid = match GroupMessageRecord::select_by_nano_id(rb, &item.nano_id.unwrap_or_default()).await? {
                Some(m) => match m.group_uuid {
                    Some(g) => g,
                    None => continue,
                },
                None => {
                    warn!("[已读桥接] 群消息不存在: {:?}", item.nano_id);
                    continue;
                }
            };
            // init 幂等且恰好把游标初始化到 max —— 单步完成"建行+读到底"
            UserSession::init_for_group_join(rb, &me_uuid, &group_uuid).await?;
            // 若行已存在(老成员), init 是 no-op, 需显式推进:
            let max_id = GroupMessageRecord::max_id_by_group(rb, &group_uuid).await?;
            UserSession::update_last_read_id(rb, &me_uuid, &group_uuid, max_id).await?;
        } else {
            // 单聊: 由用户对派生会话(不依赖 nano_id 反查, 零扫描)
            let peer_str = if item.send_user == me { item.recv_user } else { item.send_user };
            let Ok(peer) = peer_str.parse::<Uuid>() else {
                warn!("[已读桥接] 对方 uuid 非法: {}", peer_str);
                continue;
            };
            let session_uuid = common::utils::session_uuid::single_session_uuid(&me_uuid, &peer);
            // 行不存在(未聚合)则先建 0 游标行, 再推进 —— 保证已读不丢
            UserSession::upsert(rb, &UserSession {
                id: None, user_uuid: me_uuid, session_uuid,
                session_type: Some(SESSION_TYPE_SINGLE), peer_uuid: Some(peer),
                last_read_id: None, synced_id: None, pinned: None, muted: None,
                deleted_at: None, created_at: None, updated_at: None,
            }).await?;
            let max_id = ChatMessageRecord::max_id_by_session(rb, &session_uuid).await?;
            UserSession::update_last_read_id(rb, &me_uuid, &session_uuid, max_id).await?;
        }
    }
    Ok(CommonResponseNoDataRef::success_empty())
}
```

> 鉴权用 header 的 uuid（`me`），**不信 DTO 里的 `recv_user`** —— 旧路径 :668 有"reader 是收发双方之一"校验，新路径直接以登录身份为准，更严。

## 5. `get_unread_chat_record` 临时重写（修缺陷 A）

`text_msg_service.rs:36-92` 整个函数体替换 —— 响应结构（`Vec<ChatMessageRecord>`）不变，客户端零改动，但游标从"全局时间戳"改为"按会话独立"：

```rust
/// 获取未读消息(临时重写, 修复缺陷A: 全局时间戳→按会话游标)。
/// 任务05 由 /session/sync 正式取代后废弃。
/// 依赖 user_session 行存在(上线聚合建立); 无行/无未读返回空。
pub async fn get_unread_chat_record(
    rb: &RBatis,
    me: &str,
) -> Result<String, anyhow::Error> {
    let me_uuid: Uuid = me.parse()?;
    let sessions = UserSession::select_by_user(rb, &me_uuid).await?;
    let mut unread: Vec<ChatMessageRecord> = Vec::new();
    for s in sessions {
        // 只处理单聊(群聊走 /group/chat/message/unread, 与客户端现状一致)
        if s.session_type != Some(SESSION_TYPE_SINGLE) { continue; }
        let cursor = s.last_read_id.unwrap_or(0);
        let msgs = ChatMessageRecord::select_unread_by_cursor(
            rb, &s.session_uuid, &me_uuid, cursor, 500,
        ).await?;
        // 只把"我收到的"计为未读(自己发的不计, 主方案 §9.1)
        unread.extend(msgs);
    }
    if unread.is_empty() {
        return Ok(CommonResponseNoDataRef::success_empty());
    }
    Ok(CommonResponseRef::<Vec<ChatMessageRecord>>::success_json(&unread)?)
}
```

```rust
            });
        }
    }
    Ok(result)
}
```

> 依赖 `user_session` 行存在（任务 03 聚合建立）。**开发期注意**：若用户从未触发过聚合（无行），该接口返回空 —— 与客户端"无未读"处理一致（`pull_group_messages` 对空列表直接返回），不构成破坏。
>
> `GroupMember::select_groups_by_user` 若因此失去最后一个调用者，一并删除（rg 复核后决定）。
